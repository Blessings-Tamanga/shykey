# Shykey frontend

Shykey is a mobile-first React application built with Vite and TypeScript. It keeps the original prototypeâ€™s phone-width canvas, gradient backdrop, typography, mode colors, mascot, and mission-to-check-in loop while separating each screen and its reusable UI.

## Run locally

```sh
npm install
npm run dev
```

Create and verify the production build with `npm run build`, then preview it with `npm run preview`.

## Project structure

```text
src/
  components/  Shared buttons, mascot, mode cards, iconography, and screen helpers
  data/        Temporary demo content and sample readings
  hooks/       Shared demo state and theme behavior
  layouts/     Shared routed application shell
  pages/       Home, Mission, Check-in, Win, Chat, Read, and Profile screens
  services/    Mock service and typed FastAPI-compatible service adapter
  styles/      Design tokens, responsive styling, and reusable UI styles
  types/       Shared application and service contracts
  utils/       Local demo persistence and optional haptics
```

The app uses React Router paths `/`, `/mission`, `/checkin`, `/win`, `/chat`, `/read`, and `/profile`. `mockApi` is the active service, so screens work without a backend. Mock copy and example outputs live in `src/data/mockData.ts`; service contracts and the replaceable API adapter live in `src/services/`. The adapter targets documented `/v1` FastAPI endpoints and can be created with `createApiService(baseUrl, getAccessToken)` when the backend is ready. Client-side streak and mission data are demo state, not authoritative records.

## Notes

- Demo progress is stored locally under `shykey:v1`; reset it from Profile.
- The message-reading screen uses fixed illustrative readings. It does not send or persist pasted text in the mock experience.
- The UI uses the documented fixed support levels without presenting them as probabilities.
- No PWA, authentication, or live backend integration is enabled in this frontend milestone.

interface StreakWeekProps {
  streak: number;
  doneToday: boolean;
}

export function StreakWeek({ streak, doneToday }: StreakWeekProps) {
  const today = new Date().getDay();
  return (
    <div className="week" aria-label="This week's activity">
      {['s', 'm', 't', 'w', 't', 'f', 's'].map((label, day) => {
        let state = '';
        if (day === today) state = doneToday ? 'complete' : 'current';
        else if (day < today) state = day >= today - (doneToday ? streak - 1 : streak) ? 'complete' : 'missed';
        return (
          <div key={`${label}-${day}`}>
            <i className={state} aria-label={state || 'upcoming'} />
            <span>{label}</span>
          </div>
        );
      })}
    </div>
  );
}

import { createContext, useContext, useEffect, useMemo, useState, type PropsWithChildren } from 'react';
import { INITIAL_STATS, MODES } from '../data/mockData';
import { mockApi } from '../services';
import type { AppServices, CheckinOutcome, ModeKey, Mission, ProfileStats } from '../types';
import { buzz, todayKey } from '../utils/haptics';
import { clearStats, loadStats, saveStats } from '../utils/storage';

interface ShykeyContextValue {
  stats: ProfileStats;
  modeKey: ModeKey;
  smallMission: boolean;
  mission: Mission | null;
  assignmentId: string | null;
  doneToday: boolean;
  selectMode(mode: ModeKey): void;
  loadMission(small?: boolean, mode?: ModeKey): Promise<void>;
  setSmallMission(small: boolean): void;
  commitMission(): Promise<void>;
  checkIn(outcome: CheckinOutcome): Promise<void>;
  resetDemo(): void;
  services: AppServices;
}

const ShykeyContext = createContext<ShykeyContextValue | null>(null);

export function ShykeyProvider({ children }: PropsWithChildren) {
  const [stats, setStats] = useState(() => loadStats(INITIAL_STATS));
  const [modeKey, setModeKey] = useState<ModeKey>('flirting');
  const [smallMission, setSmallMission] = useState(false);
  const [mission, setMission] = useState<Mission | null>(null);
  const [assignmentId, setAssignmentId] = useState<string | null>(null);
  const [parentAssignmentId, setParentAssignmentId] = useState<string | null>(null);

  useEffect(() => saveStats(stats), [stats]);

  const value = useMemo<ShykeyContextValue>(() => ({
    stats,
    modeKey,
    smallMission,
    mission,
    assignmentId,
    doneToday: stats.date === todayKey(),
    services: mockApi,
    selectMode(mode) {
      setModeKey(mode);
      setSmallMission(false);
    },
    async loadMission(small = smallMission, selectedMode = modeKey) {
      const selected = MODES.find((item) => item.key === selectedMode) ?? MODES[2];
      const result = await mockApi.getMission(selected.key, small ? 'small' : 'full');
      setMission(result);
      setSmallMission(small);
    },
    setSmallMission,
    async commitMission() {
      buzz();
      setAssignmentId(await mockApi.commitMission(mission, parentAssignmentId));
      setParentAssignmentId(null);
    },
    async checkIn(outcome) {
      buzz();
      await mockApi.checkIn(assignmentId, outcome);
      if (outcome === 'not_yet') {
        setParentAssignmentId(assignmentId);
        setAssignmentId(null);
        return;
      }
      setStats((current) => {
        if (current.date === todayKey()) return current;
        return {
          ...current,
          streak: current.streak + 1,
          reps: current.reps + (outcome === 'done' ? 1 : 0),
          best: Math.max(current.best, current.streak + 1),
          date: todayKey(),
        };
      });
    },
    resetDemo() {
      clearStats();
      setStats(INITIAL_STATS);
      setModeKey('flirting');
      setSmallMission(false);
      setMission(null);
      setAssignmentId(null);
      setParentAssignmentId(null);
    },
  }), [stats, modeKey, smallMission, mission, assignmentId, parentAssignmentId]);

  return <ShykeyContext.Provider value={value}>{children}</ShykeyContext.Provider>;
}

export function useShykey(): ShykeyContextValue {
  const context = useContext(ShykeyContext);
  if (!context) throw new Error('useShykey must be used inside ShykeyProvider.');
  return context;
}

import { useState } from 'react';

export function useTheme() {
  const [dark, setDark] = useState(
    document.documentElement.dataset.theme
      ? document.documentElement.dataset.theme === 'dark'
      : window.matchMedia('(prefers-color-scheme: dark)').matches,
  );

  function toggleTheme() {
    const next = !dark;
    document.documentElement.dataset.theme = next ? 'dark' : 'light';
    setDark(next);
  }

  return { dark, toggleTheme };
}

import type { AppServices, ChatReply, ModeKey, Mission, ReadingResult } from '../types';

interface ApiError {
  error?: {
    message?: string;
  };
}

interface MissionResponse {
  template_id: string;
  mode_key: ModeKey;
  variant: 'full' | 'small';
  task: string;
  why: string;
}

interface AssignmentResponse {
  id: string;
}

interface CoachResponse {
  reply: { text: string };
}

interface ReadResponse {
  readings: ReadingResult['readings'];
  what_would_settle_it: string;
}

export type AccessTokenProvider = () => string | null | Promise<string | null>;

async function request<T>(
  baseUrl: string,
  path: string,
  init?: RequestInit,
  getAccessToken?: AccessTokenProvider,
): Promise<T> {
  const headers = new Headers(init?.headers);
  headers.set('Content-Type', 'application/json');
  const token = await getAccessToken?.();
  if (token) headers.set('Authorization', `Bearer ${token}`);

  const response = await fetch(`${baseUrl.replace(/\/$/, '')}${path}`, {
    ...init,
    headers,
  });

  if (!response.ok) {
    const error = await response.json().catch((): ApiError => ({})) as ApiError;
    throw new Error(error.error?.message ?? `Request failed (${response.status})`);
  }

  return response.status === 204 ? undefined as T : response.json() as Promise<T>;
}

export function createApiService(baseUrl: string, getAccessToken?: AccessTokenProvider): AppServices {
  return {
    async getMission(mode: ModeKey, variant: 'full' | 'small'): Promise<Mission> {
      const query = new URLSearchParams({ mode, variant });
      const result = await request<MissionResponse>(baseUrl, `/v1/missions/next?${query}`, undefined, getAccessToken);
      return {
        templateId: result.template_id,
        modeKey: result.mode_key,
        task: result.task,
        why: result.why,
        variant: result.variant,
      };
    },

    async commitMission(mission: Mission | null, parentAssignmentId?: string | null): Promise<string> {
      if (!mission) throw new Error('Choose a mission before committing it.');
      const result = await request<AssignmentResponse>(baseUrl, '/v1/assignments', {
        method: 'POST',
        headers: { 'Idempotency-Key': crypto.randomUUID() },
        body: JSON.stringify({
          template_id: mission.templateId,
          variant: mission.variant,
          source: 'mission',
          conversation_id: null,
          read_id: null,
          parent_assignment_id: parentAssignmentId ?? null,
          replace_open: false,
        }),
      }, getAccessToken);
      return result.id;
    },

    async checkIn(assignmentId, outcome): Promise<void> {
      if (!assignmentId) throw new Error('There is no committed mission to check in.');
      await request(baseUrl, `/v1/assignments/${assignmentId}/checkin`, {
        method: 'POST',
        headers: { 'Idempotency-Key': crypto.randomUUID() },
        body: JSON.stringify({ outcome }),
      }, getAccessToken);
    },

    async sendCoachMessage(conversationId: string | null, text: string): Promise<ChatReply> {
      let activeConversationId = conversationId;
      if (!activeConversationId) {
        const created = await request<{ conversation_id: string; opening: { text: string } }>(
          baseUrl,
          '/v1/conversations',
          { method: 'POST', body: JSON.stringify({ mode_key: 'flirting' }) },
          getAccessToken,
        );
        activeConversationId = created.conversation_id;
      }

      const result = await request<CoachResponse>(
        baseUrl,
        `/v1/conversations/${activeConversationId}/messages`,
        {
          method: 'POST',
          headers: { 'Idempotency-Key': crypto.randomUUID() },
          body: JSON.stringify({ text, action: null, user_draft: null }),
        },
        getAccessToken,
      );
      return { conversationId: activeConversationId, text: result.reply.text };
    },

    async readMessage(text: string): Promise<ReadingResult> {
      const result = await request<ReadResponse>(baseUrl, '/v1/reads', {
        method: 'POST',
        headers: { 'Idempotency-Key': crypto.randomUUID() },
        body: JSON.stringify({ text }),
      }, getAccessToken);
      return { readings: result.readings, whatWouldSettleIt: result.what_would_settle_it };
    },

    async joinWaitlist(email: string, offerId?: string): Promise<void> {
      if (!offerId) throw new Error('An offer ID is required to join the waitlist.');
      await request(baseUrl, '/v1/waitlist', {
        method: 'POST',
        headers: { 'Idempotency-Key': crypto.randomUUID() },
        body: JSON.stringify({ email, consent: true, offer_id: offerId }),
      }, getAccessToken);
    },
  };
}

import type { ProfileStats } from '../types';

const STORAGE_KEY = 'shykey:v1';

export function loadStats(fallback: ProfileStats): ProfileStats {
  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (!saved) return fallback;

    const parsed: unknown = JSON.parse(saved);
    if (!parsed || typeof parsed !== 'object') return fallback;
    const value = parsed as Partial<ProfileStats>;
    if (
      typeof value.streak !== 'number' || !Number.isInteger(value.streak) || value.streak < 0
      || typeof value.reps !== 'number' || !Number.isInteger(value.reps) || value.reps < 0
      || typeof value.best !== 'number' || !Number.isInteger(value.best) || value.best < 0
      || !(typeof value.date === 'string' || value.date === null)
    ) return fallback;

    return { streak: value.streak, reps: value.reps, best: value.best, date: value.date };
  } catch {
    return fallback;
  }
}

export function saveStats(stats: ProfileStats): void {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(stats));
  } catch {
    // Local storage can be unavailable in private browsing; the in-memory demo still works.
  }
}

export function clearStats(): void {
  try {
    localStorage.removeItem(STORAGE_KEY);
  } catch {
    // Resetting the in-memory demo still works when local storage is unavailable.
  }
}
